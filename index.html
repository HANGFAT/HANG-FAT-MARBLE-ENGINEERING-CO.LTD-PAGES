<script>
const edgeTypes = [
{ name: "普通邊", price: 25 },
{ name: "小圓邊", price: 30 },
{ name: "奇子邊", price: 50 },
{ name: "法國邊", price: 60 },
{ name: "格角", price: 40 },
{ name: "前擋", price: 35 },
{ name: "自訂邊型/手動輸入", price: 0 }
];

function addStoneRow() {
const tbody = document.getElementById('stoneBody');
const tr = document.createElement('tr');
tr.innerHTML = `
<td><input type="text" placeholder="例: 廚檯面 1" class="stone-name"></td>
<td><input type="number" value="0" min="0" class="stone-len" oninput="calculateAll()"></td>
<td><input type="number" value="0" min="0" class="stone-wid" oninput="calculateAll()"></td>
<td><input type="number" value="1" min="1" class="stone-qty" oninput="calculateAll()"></td>
<td><span class="raw-sqft" style="color: #a0aec0; font-size: 12px;">0</span> 呎 → <span class="billed-txt"><span class="billed-sqft">0</span> 呎</span></td>
<td><button class="btn btn-del" onclick="removeRow(this)">刪除</button></td>
`;
tbody.appendChild(tr);
calculateAll();
}

function addCraftRow() {
const tbody = document.getElementById('craftBody');
const tr = document.createElement('tr');

let optionsHtml = edgeTypes.map(e => `<option value="${e.price}" data-name="${e.name}">${e.name} ${e.price > 0 ? '($' + e.price + ')' : ''}</option>`).join('');

tr.innerHTML = `
<td>
<select class="craft-type-select" onchange="onEdgeTypeChange(this)">
${optionsHtml}
</select>
<input type="text" class="custom-craft-name" placeholder="請輸入加工名稱" style="display:none; margin-top:4px;">
</td>
<td><input type="number" value="0" min="0" class="craft-len" oninput="calculateAll()"></td>
<td><span class="craft-ft">0</span> 呎</td>
<td><input type="number" value="25" min="0" class="craft-price" oninput="calculateAll()"></td>
<td><button class="btn btn-del" onclick="removeRow(this)">刪除</button></td>
`;
tbody.appendChild(tr);
calculateAll();
}

function onEdgeTypeChange(selectElem) {
const row = selectElem.parentNode.parentNode;
const selectedPrice = parseFloat(selectElem.value);
const customInput = row.querySelector('.custom-craft-name');
const priceInput = row.querySelector('.craft-price');

const selectedOption = selectElem.options[selectElem.selectedIndex];
const selectedName = selectedOption.getAttribute('data-name');

if (selectedName === "自訂邊型/手動輸入") {
customInput.style.display = "block";
} else {
customInput.style.display = "none";
priceInput.value = selectedPrice;
}
calculateAll();
}

function removeRow(btn) {
const row = btn.parentNode.parentNode;
row.parentNode.removeChild(row);
calculateAll();
}

function calculateAll() {
const unit = document.getElementById('inputUnit').value;

let toMeter = 0.01;
if (unit === 'mm') toMeter = 0.001;
if (unit === 'm') toMeter = 1.0;

const stoneRows = document.querySelectorAll('#stoneBody tr');
let totalBilledSqft = 0;

stoneRows.forEach(row => {
const len = parseFloat(row.querySelector('.stone-len').value) || 0;
const wid = parseFloat(row.querySelector('.stone-wid').value) || 0;
const qty = parseFloat(row.querySelector('.stone-qty').value) || 1;

const rawSqftPerPiece = (len * toMeter) * (wid * toMeter) * 10.764;
const billedSqftPerPiece = Math.ceil(rawSqftPerPiece);
const totalItemSqft = billedSqftPerPiece * qty;

row.querySelector('.raw-sqft').innerText = (rawSqftPerPiece * qty).toFixed(2);
row.querySelector('.billed-sqft').innerText = totalItemSqft;

totalBilledSqft += totalItemSqft;
});

const stoneUnitPrice = parseFloat(document.getElementById('stoneUnitPrice').value) || 0;
const stoneTotalMoney = totalBilledSqft * stoneUnitPrice;

const craftRows = document.querySelectorAll('#craftBody tr');
let totalCraftMoney = 0;

craftRows.forEach(row => {
const craftLen = parseFloat(row.querySelector('.craft-len').value) || 0;
const craftPrice = parseFloat(row.querySelector('.craft-price').value) || 0;

const linearFeet = (craftLen * toMeter) * 3.28;
const craftItemTotal = linearFeet * craftPrice;

row.querySelector('.craft-ft').innerText = linearFeet.toFixed(2);
totalCraftMoney += craftItemTotal;
});

const extraCraftFee = parseFloat(document.getElementById('extraCraftFee').value) || 0;
totalCraftMoney += extraCraftFee;

const grandTotal = stoneTotalMoney + totalCraftMoney;

document.getElementById('totalSqftDisplay').innerText = totalBilledSqft + " 呎";
document.getElementById('stoneTotalDisplay').innerText = "$" + Math.round(stoneTotalMoney).toLocaleString('en-US');
document.getElementById('craftTotalDisplay').innerText = "$" + Math.round(craftTotalMoney).toLocaleString('en-US');
document.getElementById('grandTotalDisplay').innerText = "$" + Math.round(grandTotal).toLocaleString('en-US');
}

window.onload = function() {
addStoneRow();
addCraftRow();
};
</script>
